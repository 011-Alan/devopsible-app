#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

int n;
int count;
sem_t *forks;
pthread_t *philosophers;

void *philosopher(void *arg)
{
    int id = *(int *)arg;
    int left = id;
    int right = (id + 1) % n;

    for(int i=0;i<count;i++){

    printf("Philosopher %d is THINKING\n", id + 1);
    sleep(1);

    printf("Philosopher %d is HUNGRY\n", id + 1);

    /* Take forks */
    sem_wait(&forks[left]);
    printf("Philosopher %d picked fork %d\n", id + 1, left + 1);

    sem_wait(&forks[right]);
    printf("Philosopher %d picked fork %d\n", id + 1, right + 1);

    printf("Philosopher %d is EATING\n", id + 1);
    sleep(2);

    /* Put forks back */
    sem_post(&forks[right]);
    sem_post(&forks[left]);

    printf("Philosopher %d finished EATING\n", id + 1);
    }

    return NULL;
}

int main()
{
    printf("Enter number of philosophers: ");
    scanf("%d", &n);

    printf("Enter number turns: ");
    scanf("%d", &count);

    if (n < 2) {
        printf("Number of philosophers must be at least 2.\n");
        return 0;
    }

    forks = malloc(n * sizeof(sem_t));
    philosophers = malloc(n * sizeof(pthread_t));

    int id[n];

    for (int i = 0; i < n; i++)
        sem_init(&forks[i], 0, 1);

    for (int i = 0; i < n; i++) {
        id[i] = i;
        pthread_create(&philosophers[i], NULL, philosopher, &id[i]);
    }

    for (int i = 0; i < n; i++)
        pthread_join(philosophers[i], NULL);

    for (int i = 0; i < n; i++)
        sem_destroy(&forks[i]);

    free(forks);
    free(philosophers);

    return 0;
}

int data = 0;
int read_count = 0;

sem_t mutex;
sem_t rw;

void *reader(void *arg)
{
    int id = *(int *)arg;

    sem_wait(&mutex);

    read_count++;

    if (read_count == 1)
        sem_wait(&rw);

    sem_post(&mutex);

    printf("Reader %d is READING data = %d\n", id, data);
    sleep(1);

    sem_wait(&mutex);

    read_count--;

    if (read_count == 0)
        sem_post(&rw);

    sem_post(&mutex);

    printf("Reader %d finished reading\n", id);

    return NULL;
}

void *writer(void *arg)
{
    int id = *(int *)arg;

    sem_wait(&rw);

    data++;

    printf("Writer %d is WRITING data = %d\n", id, data);
    sleep(2);

    sem_post(&rw);

    printf("Writer %d finished writing\n", id);

    return NULL;
}

int main()
{
    int nr, nw;

    printf("Enter number of readers: ");
    scanf("%d", &nr);

    printf("Enter number of writers: ");
    scanf("%d", &nw);

    if (nr <= 0 || nw <= 0) {
        printf("Invalid input.\n");
        return 0;
    }

    pthread_t readers[nr];
    pthread_t writers[nw];

    int rid[nr];
    int wid[nw];

    sem_init(&mutex, 0, 1);
    sem_init(&rw, 0, 1);

    for (int i = 0; i < nr; i++) {
        rid[i] = i + 1;
        pthread_create(&readers[i], NULL, reader, &rid[i]);
    }

    for (int i = 0; i < nw; i++) {
        wid[i] = i + 1;
        pthread_create(&writers[i], NULL, writer, &wid[i]);
    }

    for (int i = 0; i < nr; i++)
        pthread_join(readers[i], NULL);

    for (int i = 0; i < nw; i++)
        pthread_join(writers[i], NULL);

    sem_destroy(&mutex);
    sem_destroy(&rw);

    return 0;
}

int buffer[100];
int size;

int in = 0;
int out = 0;

sem_t empty;
sem_t full;
sem_t mutex;

void *producer(void *arg)
{
    int items;

    printf("Enter number of items to produce: ");
    scanf("%d", &items);

    for (int i = 1; i <= items; i++) {

        sem_wait(&empty);
        sem_wait(&mutex);

        buffer[in] = i;
        printf("Producer produced: %d\n", i);

        in = (in + 1) % size;

        sem_post(&mutex);
        sem_post(&full);

        sleep(1);
    }

    return NULL;
}

void *consumer(void *arg)
{
    int items;

    printf("Enter number of items to consume: ");
    scanf("%d", &items);

    for (int i = 0; i < items; i++) {

        sem_wait(&full);
        sem_wait(&mutex);

        int item = buffer[out];

        printf("Consumer consumed: %d\n", item);

        out = (out + 1) % size;

        sem_post(&mutex);
        sem_post(&empty);

        sleep(1);
    }

    return NULL;
}

int main()
{
    pthread_t p, c;

    printf("Enter buffer size: ");
    scanf("%d", &size);

    if (size <= 0 || size > 100) {
        printf("Invalid buffer size.\n");
        return 0;
    }

    sem_init(&empty, 0, size);
    sem_init(&full, 0, 0);
    sem_init(&mutex, 0, 1);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    pthread_join(p, NULL);
    pthread_join(c, NULL);

    sem_destroy(&empty);
    sem_destroy(&full);
    sem_destroy(&mutex);

    return 0;
}

        scanf("%d",&n);

        struct Process p[n];

        for(int i=0;i<n;i++){
                p[i].pid = i + 1;

                printf("\nEnter the AT for P%d:",i+1);
                scanf("%d",&p[i].at);

                printf("\nEnter the BT for P%d:",i+1);
                scanf("%d",&p[i].bt);
        }

        for(int i=0;i<n-1;i++){
                for(int j=i+1;j<n;j++){
                        if(p[i].at>p[j].at){
                                struct Process temp = p[i];
                                p[i] = p[j];
                                p[j] = temp;
                        }
                }
        }

        int time = 0;
        int k=0;
        int arr[100];

#include <stdio.h>
#define MAX 10

int P, R, count = 0;
int a[MAX][MAX], m[MAX][MAX], n[MAX][MAX], av[MAX];

int safe(int work[], int finish[])
{
    int done = 0;

    while (done < P)
    {
        int found = 0;

        for (int i = 0; i < P; i++)
        {
            if (!finish[i])
            {
                int ok = 1;

                for (int j = 0; j < R; j++)
                    if (n[i][j] > work[j])
                        ok = 0;

                if (ok)
                {
                    for (int j = 0; j < R; j++)
                        work[j] += a[i][j];

                    finish[i] = 1;
                    done++;
                    found = 1;
                }
            }
        }

        if (!found)
            return 0;
    }

    return 1;
}

void sequences(int work[], int finish[], int s[], int level)
{
    if (level == P)
    {
        printf("%d. ", ++count);

        for (int i = 0; i < P; i++)
            printf("P%d%s", s[i], i == P - 1 ? "\n" : " -> ");

        return;
    }

    for (int i = 0; i < P; i++)
    {
        int ok = 1;

        for (int j = 0; j < R; j++)
            if (n[i][j] > work[j])
                ok = 0;

        if (!finish[i] && ok)
        {
            finish[i] = 1;
            s[level] = i;

            for (int j = 0; j < R; j++)
                work[j] += a[i][j];

            sequences(work, finish, s, level + 1);

            for (int j = 0; j < R; j++)
                work[j] -= a[i][j];

            finish[i] = 0;
        }
    }
}
